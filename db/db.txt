CREATE TABLE IF NOT EXISTS giorno (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,                    -- Giorno della settimana (Lunedì, Martedì, etc.)
  lavorativo_mattino BOOLEAN NOT NULL,   -- Indica se il giorno è lavorativo al mattino
  lavorativo_pomeriggio BOOLEAN NOT NULL,-- Indica se il giorno è lavorativo al pomeriggio
  data DATE NOT NULL,                    -- Data specifica del giorno
  orario_mattino TEXT DEFAULT '09:00-13:00', -- Orario mattino con valore di default
  orario_pomeriggio TEXT DEFAULT '15:00-20:00' -- Orario pomeriggio con valore di default
);
CREATE TABLE users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nome TEXT NOT NULL,
  cognome TEXT NOT NULL,
  nomeutente TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL,
  created TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  active BOOLEAN DEFAULT 1
);
